import {
  IsArray,
  IsOptional,
  IsString,
  IsEnum,
  IsBoolean,
  IsInt,
  Min,
  Max,
  ArrayUnique,
} from 'class-validator';

import { Transform, Type } from 'class-transformer';
import { MembershipRole } from '@prisma/client';



export class QueryOrganizationMembersDto {
  @IsOptional()
  @Transform(({ value }) => {
    if (Array.isArray(value)) {
      return value;
    }

    if (typeof value === 'string') {
      return value.split(',').filter(Boolean);
    }

    return value;
  })
  @IsArray()
  @ArrayUnique()
  @IsString({ each: true })
  restaurantIds?: string[];

  @IsOptional()
  @IsEnum(MembershipRole)
  role?: MembershipRole;

  @IsOptional()
  @Transform(({ value }) => {
    if (value === 'true') return true;
    if (value === 'false') return false;
    return value;
  })
  @IsBoolean()
  allRestaurants?: boolean;

  @IsOptional()
  @Type(() => Number)
  @IsInt()
  @Min(1)
  page = 1;

  @IsOptional()
  @Type(() => Number)
  @IsInt()
  @Min(1)
  @Max(100)
  limit = 20;
}

    /**
     * Obtiene el identificador del usuario autenticado.
     */
    private getAuthenticatedUserId(
        req: AuthenticatedRequest,
    ): string {
        const userId = req.user?.userId;

        if (!userId) {
            throw new UnauthorizedException(
                'Authenticated user ID not found',
            );
        }

        return userId;
    }

     * GET /api/organizations/:organizationId/members
     * Lista los miembros de una organización.

        @Req() req: AuthenticatedRequest,
        @Param('organizationId') organizationId: string,
        @Query() query: QueryOrganizationMembersDto,
    ) {
        return this.organizationMembersService.findAll(
            this.getAuthenticatedUserId(req),
            organizationId,
            query,
        );
    }

    findOne(

import {
  Injectable,
  ForbiddenException,
  NotFoundException,
  ConflictException,
  BadRequestException,
} from '@nestjs/common';

import * as argon2 from 'argon2';

import { Prisma, type PrismaClient } from '@prisma/client';

import { PrismaService } from '../../../database/prisma/prisma.service.js';

import {
  QueryOrganizationMembersDto,
} from '../dto/query-organization-members.dto/query-organization-members.dto.js';

import {
  CreateOrganizationMemberDto,
} from '../dto/create-organization-member.dto/create-organization-member.dto.js';

import {
  UpdateOrganizationMemberDto,
} from '../dto/update-organization-member.dto/update-organization-member.dto.js';

import {
  AssignRestaurantsDto,
} from '../dto/assign-restaurants.dto/assign-restaurants.dto.js';

import {
  PlanLimitsService,
} from '../../plan-access/application/services/plan-limits.service.js';

import {
  FeatureAccessService,
} from '../../plan-access/application/services/feature-access.service.js';

@Injectable()
export class OrganizationMembersService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly planLimitsService: PlanLimitsService,
    private readonly featureAccessService: FeatureAccessService,
  ) { }

  // =====================================================
  // AUTORIZACIÓN
  // =====================================================

  private async assertCanManageMembers(
    actorId: string,
    organizationId: string,
  ) {
    const membership =
      await this.prisma.organizationMember.findUnique({
        where: {
          userId_organizationId: {
            userId: actorId,
            organizationId,
          },
        },
        select: {
          id: true,
          role: true,
          user: {
            select: {
              status: true,
            },
          },
          organization: {
            select: {
              status: true,
            },
          },
        },
      });

    if (
      !membership ||
      membership.user.status !== 'ACTIVE' ||
      membership.organization.status !== 'ACTIVE'
    ) {
      throw new ForbiddenException(
        'You cannot manage members of this organization',
      );
    }

    if (
      membership.role !== 'OWNER' &&
      membership.role !== 'ADMIN'
    ) {
      throw new ForbiddenException(
        'You do not have permission to manage organization members',
      );
    }

    return membership;
  }

  // =====================================================
  // VALIDACIÓN DE FUNCIONALIDAD DEL PLAN
  // =====================================================

  private async assertTeamMembersFeature(
    organizationId: string,
  ): Promise<void> {
    await this.featureAccessService.assertFeatureEnabled(
      organizationId,
      'TEAM_MEMBERS',
    );
  }

  // =====================================================
  // VALIDACIÓN DE RESTAURANTES
  // =====================================================

  private async assertRestaurantsBelongToOrganization(
    organizationId: string,
    restaurantIds: string[],
  ): Promise<string[]> {
    const uniqueIds = [...new Set(restaurantIds)];

    if (uniqueIds.length === 0) {
      throw new BadRequestException(
        'At least one restaurant must be selected',
      );
    }

    const restaurants =
      await this.prisma.restaurant.findMany({
        where: {
          id: {
            in: uniqueIds,
          },
          organizationId,
          status: 'ACTIVE',
        },
        select: {
          id: true,
        },
      });

    if (restaurants.length !== uniqueIds.length) {
      throw new BadRequestException(
        'One or more restaurants are invalid, inactive, or belong to another organization',
      );
    }

    return uniqueIds;
  }

  // =====================================================
  // LISTAR MIEMBROS
  // =====================================================

  async findAll(
    actorId: string,
    organizationId: string,
    query: QueryOrganizationMembersDto,
  ) {
    await this.assertCanManageMembers(
      actorId,
      organizationId,
    );

    const {
      restaurantIds,
      role,
      allRestaurants,
      page = 1,
      limit = 20,
    } = query;

    let assignedUserIds: string[] = [];

    if (restaurantIds?.length) {
      const validRestaurantIds =
        await this.assertRestaurantsBelongToOrganization(
          organizationId,
          restaurantIds,
        );

      const assignments =
        await this.prisma.restaurantMember.findMany({
          where: {
            restaurantId: {
              in: validRestaurantIds,
            },
            restaurant: {
              organizationId,
            },
          },
          select: {
            userId: true,
          },
          distinct: ['userId'],
        });

      assignedUserIds = assignments.map(
        (assignment) => assignment.userId,
      );
    }

    const where: Prisma.OrganizationMemberWhereInput = {
      organizationId,

      ...(role !== undefined
        ? { role }
        : {}),

      ...(allRestaurants !== undefined
        ? { allRestaurants }
        : {}),

      ...(restaurantIds?.length
        ? {
          OR: [
            {
              allRestaurants: true,
            },
            {
              userId: {
                in: assignedUserIds,
              },
            },
          ],
        }
        : {}),
    };

    const [total, members] =
      await this.prisma.$transaction([
        this.prisma.organizationMember.count({
          where,
        }),

        this.prisma.organizationMember.findMany({
          where,
          skip: (page - 1) * limit,
          take: limit,
          orderBy: {
            createdAt: 'desc',
          },
          select: {
            id: true,
            userId: true,
            role: true,
            allRestaurants: true,
            createdAt: true,
            updatedAt: true,

            user: {
              select: {
                id: true,
                email: true,
                firstName: true,
                status: true,
              },
            },
          },
        }),
      ]);

    const memberIds = members.map(
      (member) => member.userId,
    );

    const restaurantMemberships =
      memberIds.length > 0
        ? await this.prisma.restaurantMember.findMany({
          where: {
            userId: {
              in: memberIds,
            },
            restaurant: {
              organizationId,
            },
          },
          select: {
            userId: true,
            restaurantId: true,
            role: true,
            roleId: true,

            restaurant: {
              select: {
                name: true,
                status: true,
              },
            },
          },
        })
        : [];

    const assignmentsByUser = new Map<
      string,
      typeof restaurantMemberships
    >();

    for (const assignment of restaurantMemberships) {
      const current =
        assignmentsByUser.get(assignment.userId) ?? [];

      current.push(assignment);

      assignmentsByUser.set(
        assignment.userId,
        current,
      );
    }

    const roles = await this.prisma.role.findMany({
      select: {
        code: true,
        name: true,
      },
    });

    const roleNames = new Map(
      roles.map((role) => [role.code, role.name]),
    );

    return {
      data: members.map((member) => ({
        id: member.id,
        userId: member.userId,
        role: member.role,
        roleName: roleNames.get(member.role) ?? member.role,
        allRestaurants: member.allRestaurants,
        createdAt: member.createdAt,
        updatedAt: member.updatedAt,

        user: member.user,

        restaurants: member.allRestaurants
          ? []
          : (
            assignmentsByUser.get(member.userId) ?? []
          ).map((assignment) => ({
            id: assignment.restaurantId,
            name: assignment.restaurant.name,
            status: assignment.restaurant.status,
            role: assignment.role,
            roleId: assignment.roleId,
          })),
      })),

      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  // =====================================================
  // CONSULTAR UN MIEMBRO
  // =====================================================

  async findOne(
    actorId: string,
    organizationId: string,
    memberId: string,
  ) {
    await this.assertCanManageMembers(
      actorId,
      organizationId,
    );

    const member =
      await this.prisma.organizationMember.findFirst({
        where: {
          id: memberId,
          organizationId,
        },
        select: {
          id: true,
          userId: true,
          role: true,
          allRestaurants: true,
          createdAt: true,
          updatedAt: true,

          user: {
            select: {
              id: true,
              email: true,
              firstName: true,
              status: true,
            },
          },
        },
      });

    if (!member) {
      throw new NotFoundException(
        'Organization member not found',
      );
    }

    const assignments =
      await this.prisma.restaurantMember.findMany({
        where: {
          userId: member.userId,
          restaurant: {
            organizationId,
          },
        },
        select: {
          id: true,
          restaurantId: true,
          role: true,
          roleId: true,

          restaurant: {
            select: {
              name: true,
              status: true,
            },
          },
        },
      });

    return {
      ...member,

      restaurants: member.allRestaurants
        ? []
        : assignments.map((assignment) => ({
          id: assignment.restaurantId,
          name: assignment.restaurant.name,
          status: assignment.restaurant.status,
          role: assignment.role,
          roleId: assignment.roleId,
        })),
    };
  }

  // =====================================================
  // INCORPORAR UN USUARIO EXISTENTE
  // =====================================================

  async create(
    actorId: string,
    organizationId: string,
    dto: CreateOrganizationMemberDto,
  ) {
    await this.assertCanManageMembers(
      actorId,
      organizationId,
    );

    await this.assertTeamMembersFeature(organizationId);

    if (dto.role === 'OWNER') {
      throw new BadRequestException(
        'Use the ownership transfer process to assign the OWNER role',
      );
    }

    if (
      !dto.allRestaurants &&
      (!dto.restaurantIds || dto.restaurantIds.length === 0)
    ) {
      throw new BadRequestException(
        'Select at least one restaurant or enable access to all restaurants',
      );
    }

    const email = dto.email.trim().toLowerCase();

    const restaurantIds = dto.allRestaurants
      ? []
      : await this.assertRestaurantsBelongToOrganization(
        organizationId,
        dto.restaurantIds ?? [],
      );

    const existingUser = await this.prisma.user.findUnique({
      where: {
        email,
      },
      select: {
        id: true,
        status: true,
      },
    });

    if (
      existingUser &&
      existingUser.status !== 'ACTIVE'
    ) {
      throw new BadRequestException(
        'This user account is not active',
      );
    }

    if (!existingUser && !dto.password) {
      throw new BadRequestException(
        'A password is required when creating a new user',
      );
    }

    const passwordHash = existingUser
      ? undefined
      : await argon2.hash(dto.password!, {
        type: argon2.argon2id,
      });

    try {
      return await this.prisma.$transaction(
        async (tx) => {
          /*
           * Serializa las operaciones que intenten modificar
           * los miembros de esta misma organización, siempre
           * que utilicen este mismo protocolo de bloqueo.
           */
          const organizations = await tx.$queryRaw<
            Array<{ id: string }>
          >`
          SELECT id
          FROM organizations
          WHERE id = ${organizationId}
          FOR UPDATE
        `;

          if (organizations.length === 0) {
            throw new NotFoundException(
              'Organization not found',
            );
          }

          /*
           * Volver a comprobar la membresía del actor dentro
           * de la transacción evita depender exclusivamente
           * de la validación previa.
           */
          const actorMembership =
            await tx.organizationMember.findUnique({
              where: {
                userId_organizationId: {
                  userId: actorId,
                  organizationId,
                },
              },
              select: {
                role: true,
                user: {
                  select: {
                    status: true,
                  },
                },
                organization: {
                  select: {
                    status: true,
                  },
                },
              },
            });

          if (
            !actorMembership ||
            actorMembership.user.status !== 'ACTIVE' ||
            actorMembership.organization.status !== 'ACTIVE' ||
            !['OWNER', 'ADMIN'].includes(
              actorMembership.role,
            )
          ) {
            throw new ForbiddenException(
              'You cannot manage members of this organization',
            );
          }

          /*
           * Comprobar el plan y el cupo dentro de la misma
           * transacción que crea la membresía.
           */
          const plan =
            await this.planLimitsService.getOrganizationPlan(
              organizationId,
            );

          const currentUsers =
            await tx.organizationMember.count({
              where: {
                organizationId,
              },
            });

          if (
            plan.limits.maxUsers !== null &&
            currentUsers >= plan.limits.maxUsers
          ) {
            throw new ForbiddenException({
              code: 'PLAN_LIMIT_REACHED',
              message:
                'La organización alcanzó el límite de usuarios de su plan.',
              limit: 'maxUsers',
              maximum: plan.limits.maxUsers,
              current: currentUsers,
              plan: plan.planCode,
            });
          }

          /*
           * Resolver el usuario dentro de la transacción.
           * La restricción única de email sigue siendo la
           * garantía definitiva contra duplicados.
           */
          const user = existingUser
            ? await tx.user.findUnique({
              where: {
                id: existingUser.id,
              },
              select: {
                id: true,
                status: true,
              },
            })
            : await tx.user.create({
              data: {
                email,
                firstName: dto.firstName.trim(),
                passwordHash: passwordHash!,
                status: 'ACTIVE',
              },
              select: {
                id: true,
                status: true,
              },
            });

          if (!user || user.status !== 'ACTIVE') {
            throw new BadRequestException(
              'This user account is not active',
            );
          }

          const existingMembership =
            await tx.organizationMember.findUnique({
              where: {
                userId_organizationId: {
                  userId: user.id,
                  organizationId,
                },
              },
              select: {
                id: true,
              },
            });

          if (existingMembership) {
            throw new ConflictException(
              'This user already belongs to the organization',
            );
          }

          const member =
            await tx.organizationMember.create({
              data: {
                userId: user.id,
                organizationId,
                role: dto.role,
                allRestaurants: dto.allRestaurants,
              },
              select: {
                id: true,
                userId: true,
                organizationId: true,
                role: true,
                allRestaurants: true,
                createdAt: true,
              },
            });

          if (restaurantIds.length > 0) {
            await tx.restaurantMember.createMany({
              data: restaurantIds.map((restaurantId) => ({
                userId: user.id,
                restaurantId,
                role: dto.role,
              })),

          timeout: 10000,

        error instanceof Prisma.PrismaClientKnownRequestError &&
        error.code === 'P2002'
      ) {
        throw new ConflictException(
          'This user or membership already exists',
        );
      }

      throw error;

      await this.prisma.organizationMember.findFirst({

          role: true,
          allRestaurants: true,
        },

      );
    }

    if (
      member.role === 'OWNER' &&
      actor.role !== 'OWNER'
    ) {
      throw new ForbiddenException(
        'Only the organization owner can modify the owner membership',
      );
    }

    if (dto.role === 'OWNER') {
      throw new BadRequestException(

      );
    }

      );

      );

          },

              }
              : {}),

          },

            organizationId: true,
            role: true,
            allRestaurants: true,
            updatedAt: true,
          },
        });

            type: argon2.argon2id,
          },
        );

        await tx.user.update({
          where: {
            id: member.userId,
          },
          data: {

          },
        });
      }

      if (nextAllRestaurants) {
        await tx.restaurantMember.deleteMany({

            userId: member.userId,
            restaurant: {

            },
          },
        });
      } else if (restaurantIds !== undefined) {

          where: {
            userId: member.userId,
            restaurant: {

            },
          },
        });

            restaurantId,
            role: dto.role ?? member.role,
          })),
        });
      } else if (dto.role !== undefined) {
        await tx.restaurantMember.updateMany({

            userId: member.userId,
            restaurant: {
              organizationId,
            },
          },
          data: {
            role: dto.role,
          },
        });
      }

    });
  }

    };
  }

  Injectable,
} from '@nestjs/common';

import { PrismaService } from '../../../database/prisma/prisma.service.js';

import {
  SYSTEM_ROLE_PERMISSIONS,
} from '../constants/role-permissions.constant.js';

type SystemRole = keyof typeof SYSTEM_ROLE_PERMISSIONS;

@Injectable()
export class PermissionResolverService {
  constructor(
    private readonly prisma: PrismaService,
  ) {}

  /**
   * Resuelve los permisos de una membresía de restaurante.
   *
   * Si existe un rol personalizado, utiliza exclusivamente
   * los permisos de ese rol.
   *
   * Si no existe un rol personalizado, utiliza los permisos
   * definidos para el rol del sistema.
   */
  async resolveRestaurantPermissions(
    restaurantMemberId: string,
    role: string,
    customRoleId: string | null,
  ): Promise<string[]> {
    if (customRoleId) {
      const customRole = await this.prisma.role.findFirst({
        where: {
          id: customRoleId,
          isActive: true,
          restaurant: {
            memberships: {
              some: {
                id: restaurantMemberId,
              },
            },
          },
        },
        select: {
          permissions: {
            where: {
              permission: {
                isActive: true,
              },
            },
            select: {
              permission: {
                select: {
                  code: true,
                },
              },
            },
          },
        },
      });

      if (!customRole) {
        throw new ForbiddenException({
          code: 'INVALID_CUSTOM_ROLE',
          message: 'The assigned custom role is invalid or inactive.',
        });
      }

      return [
        ...new Set(
          customRole.permissions.map(
            ({ permission }) => permission.code,
          ),
        ),
      ];
    }

    return this.resolveSystemRolePermissions(role);
  }

  /**
   * Resuelve los permisos de un rol del sistema.
   *
   * Los roles desconocidos no reciben permisos.
   */
  resolveSystemRolePermissions(
    role: string,
  ): string[] {
    if (
      !Object.prototype.hasOwnProperty.call(
        SYSTEM_ROLE_PERMISSIONS,
        role,
      )
    ) {
      throw new ForbiddenException({
        code: 'INVALID_SYSTEM_ROLE',
        message: 'The assigned system role is invalid.',
      });
    }

    const permissions =
      SYSTEM_ROLE_PERMISSIONS[role as SystemRole];

    return [...new Set(permissions)];
  }

export interface WorkspaceAccessContext {

  organizationId: string;

@Injectable()
export class WorkspaceAuthorizationService {
  constructor(
    private readonly prisma: PrismaService,
  ) {}

  /**
   * Devuelve el contexto autorizado del workspace.
   *
   * La organización se obtiene del restaurante validado,
   * nunca de un identificador proporcionado por el cliente.
   */
  async getWorkspaceAccess(
    userId: string,
    restaurantId: string,
  ): Promise<WorkspaceAccessContext> {
    const now = new Date();

    // 1. Validar restaurante y organización.
    const restaurant = await this.prisma.restaurant.findFirst({
      where: {
        id: restaurantId,
        status: 'ACTIVE',
        organization: {
          status: 'ACTIVE',
        },
      },
      select: {
        id: true,
        organizationId: true,
      },
    });

    if (!restaurant) {
      throw new NotFoundException(
        'Active restaurant not found',
      );
    }

    // 2. Validar usuario.
    const user = await this.prisma.user.findUnique({
      where: {
        id: userId,
      },
      select: {
        id: true,
        status: true,
      },
    });

    if (!user || user.status !== 'ACTIVE') {
      throw new UnauthorizedException(
        'User is not authenticated',
      );
    }

    // 3. Validar membresía de organización.
    const organizationMembership =
      await this.prisma.organizationMember.findFirst({
        where: {
          userId,
          organizationId: restaurant.organizationId,
        },
        select: {
          id: true,
          role: true,
          allRestaurants: true,
        },
      });

    if (!organizationMembership) {
      throw new NotFoundException(
        'Active workspace membership not found',
      );
    }

    // 4. Buscar membresía del restaurante.
    const restaurantMembership =
      await this.prisma.restaurantMember.findUnique({
        where: {
          userId_restaurantId: {
            userId,
            restaurantId,
          },
        },
        select: {
          id: true,
          role: true,
          customRole: {
            select: {
              id: true,
              code: true,
              name: true,
              isActive: true,
              permissions: {
                where: {
                  permission: {
                    isActive: true,
                  },
                },
                select: {
                  permission: {
                    select: {
                      code: true,
                    },
                  },
                },
              },
            },
          },
        },
      });

    if (
      !organizationMembership.allRestaurants &&
      !restaurantMembership
    ) {
      throw new NotFoundException(
        'Active workspace membership not found',
      );
    }

    // 5. Determinar el rol efectivo.
    const effectiveRole =
      restaurantMembership?.role ??
      organizationMembership.role;

    const customRole =
      restaurantMembership?.customRole ?? null;

    // 6. Validar la suscripción de toda la organización.
    const subscription =
      await this.prisma.subscription.findFirst({
        where: {
          organizationId: restaurant.organizationId,
          status: 'ACTIVE',
          currentPeriodEnd: {
            gt: now,
          },
        },
        orderBy: {
          currentPeriodEnd: 'desc',
        },
        select: {
          id: true,
          status: true,
          currentPeriodEnd: true,
        },
      });

    if (!subscription) {
      const latestSubscription =
        await this.prisma.subscription.findFirst({
          where: {
            organizationId: restaurant.organizationId,
          },
          orderBy: {
            createdAt: 'desc',
          },
          select: {
            id: true,
            status: true,
            planId: true,
            plan: {
              select: {
                id: true,
                name: true,
                code: true,
              },
            },
          },
        });

      throw new ForbiddenException({
        code: 'SUBSCRIPTION_REQUIRED',
        subscriptionId: latestSubscription?.id ?? null,
        subscriptionStatus: latestSubscription?.status ?? null,
        previousPlan: latestSubscription?.plan ?? null,
        message:
          'An active subscription is required to access this workspace.',
      });
    }

    // 7. Resolver permisos efectivos.
    let permissions: string[];

    if (customRole) {
      if (!customRole.isActive) {
        throw new ForbiddenException(
          'The assigned custom role is inactive',
        );
      }

      permissions = [
        ...new Set(
          customRole.permissions.map(
            ({ permission }) => permission.code,
          ),
        ),
      ];
    } else {
      const rolePermissions =
        SYSTEM_ROLE_PERMISSIONS[effectiveRole];

      if (!rolePermissions) {
        throw new ForbiddenException(
          'The assigned role has no permission configuration',
        );
      }

      permissions = [...new Set(rolePermissions)];
    }

    return {
      restaurantId: restaurant.id,
      organizationId: restaurant.organizationId,
      permissions,
    };
  }

import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import type { Request } from 'express';

import { REQUIRED_PERMISSIONS_KEY } from '../decorators/require-permissions.decorator.js';

interface AuthenticatedUser {
  userId: string;
  email: string;
}

interface WorkspaceAuthorization {
  permissions: string[];
}

interface AuthorizedRequest extends Request {
  user?: AuthenticatedUser;
  workspaceAuthorization?: WorkspaceAuthorization;
}

@Injectable()
export class PermissionsGuard implements CanActivate {
  constructor(private readonly reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredPermissions =
      this.reflector.getAllAndOverride<string[]>(
        REQUIRED_PERMISSIONS_KEY,
        [context.getHandler(), context.getClass()],
      );

    // Los endpoints sin permisos declarados no requieren este guard.
    if (!requiredPermissions?.length) {
      return true;
    }

    const request = context
      .switchToHttp()
      .getRequest<AuthorizedRequest>();

    if (!request.user?.userId) {
      throw new UnauthorizedException(
        'User is not authenticated',
      );
    }

    const workspace = request.workspaceAuthorization;

    if (!workspace) {
      throw new ForbiddenException(
        'Workspace authorization context is missing',
      );
    }

    const hasAllPermissions = requiredPermissions.every(
      (permission) => workspace.permissions.includes(permission),
    );

    if (!hasAllPermissions) {
      throw new ForbiddenException(
        'You do not have the required permissions',
      );
    }

    return true;
  }

interface JwtPayload {

  email: string;
}

@Injectable()
export class JwtStrategy extends PassportStrategy(
  Strategy,
  'jwt',
) {
  constructor(
    configService: ConfigService,
    private readonly prisma: PrismaService,
  ) {
    super({
      jwtFromRequest: ExtractJwt.fromExtractors([
        (request: Request) =>
          request?.cookies?.access_token ?? null,
      ]),

      ignoreExpiration: false,

      secretOrKey: configService.getOrThrow<string>(
        'JWT_ACCESS_SECRET',
      ),

      algorithms: ['HS256'],
    });
  }

  async validate(
    payload: JwtPayload,
  ): Promise<{
    userId: string;
    email: string;
  }> {
    if (
      !payload ||
      typeof payload.sub !== 'string' ||
      payload.sub.trim().length === 0 ||
      typeof payload.email !== 'string' ||
      payload.email.trim().length === 0
    ) {
      throw new UnauthorizedException(
        'Invalid authentication token',
      );
    }

    const user = await this.prisma.user.findUnique({
      where: {
        id: payload.sub,
      },

      select: {
        id: true,
        email: true,
        status: true,
      },
    });

    if (
      !user ||
      user.status !== 'ACTIVE' ||
      user.email !== payload.email
    ) {
      throw new UnauthorizedException(
        'Invalid authentication token',
      );
    }

    return {
      userId: user.id,
      email: user.email,
    };
  }

import helmet from 'helmet';
import cookieParser from 'cookie-parser';

import { AppModule } from './app.module.js';
import { HttpExceptionFilter } from './common/filters/http-exception.filter.js';
import { loggerConfig } from './common/logger/logger.config.js';

async function bootstrap() {
  const app = await NestFactory.create(AppModule, {

    logger: loggerConfig,

  });

  const configService =
    app.get(ConfigService);

  app.use(cookieParser());

  app.use(helmet());

  app.enableCors({
    origin:
      configService.getOrThrow<string>(
        'FRONTEND_URL',
      ),
    methods: [
      'GET',
      'HEAD',
      'POST',
      'PUT',
      'PATCH',
      'DELETE',
    ],
    credentials: true,
  });

  app.setGlobalPrefix('api');

  app.enableVersioning({
    type: VersioningType.URI,
  });

  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,
      forbidNonWhitelisted: true,
      transform: true,
    }),
  );

  app.useGlobalFilters(
    new HttpExceptionFilter(),
  );

  const swaggerConfig =
    new DocumentBuilder()
      .setTitle('PromoApp API')
      .setDescription(
        'API REST para la plataforma PromoApp',
      )
      .setVersion('1.0')
      .build();

  const swaggerDocument =
    SwaggerModule.createDocument(
      app,
      swaggerConfig,
    );

  SwaggerModule.setup(
    'api/docs',
    app,
    swaggerDocument,
  );

  await app.listen(
    configService.get<number>(
      'PORT',
      3000,
    ),
  );
}
